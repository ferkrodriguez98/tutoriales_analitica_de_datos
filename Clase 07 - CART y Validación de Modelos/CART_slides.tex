\begin{frame}{¿Qué es CART?}
    \begin{center}
        \Large CART es una técnica de aprendizaje automático para clasificación y regresión
    \end{center}
    
    \vspace{1em}
    
    \begin{itemize}
        \item Su objetivo es dividir un conjunto de datos en grupos homogéneos
        \item Utiliza el índice de Gini para evaluar la calidad de las divisiones
        \item Es un algoritmo \textbf{greedy} (voraz)
    \end{itemize}
\end{frame}

\begin{frame}{Índice de Gini}
    \begin{center}
        \Large $\mathrm{Gini} = 1 - \sum_{k=1}^K p_k^2$
    \end{center}
    
    \vspace{1em}
    
    donde $p_k$ es la proporción de la clase $k$ en el nodo
    
    \vspace{1em}
    
    \begin{itemize}
        \item \textbf{Gini = 0}: Nodo puro (todas las instancias de una clase)
        \item \textbf{Gini = 0.5}: Máxima impureza (distribución uniforme)
    \end{itemize}
\end{frame}

\begin{frame}{Algoritmo Greedy}
    ¿Por qué se dice \textbf{greedy}? Porque en cada paso tomamos la mejor decisión localmente, sin importar lo que viene después.
    
    \vspace{1em}
    
    \begin{itemize}
        \item Chequeamos cada punto de corte
        \item Determinamos si es útil o no sin tener en cuenta lo que viene después
        \item Puede llevar a soluciones subóptimas, pero es suficiente en muchos casos
    \end{itemize}
\end{frame}

\begin{frame}{Partición por Edad - Punto de Corte 23.5}
    \begin{itemize}
        \item \textbf{Edad $\leq$ 23.5}: 1 instancia (0 aprobados, 1 no aprobado)
        \item \textbf{Edad $>$ 23.5}: 4 instancias (3 aprobados, 1 no aprobado)
    \end{itemize}
    
    \vspace{0.5em}
    
    \begin{align*}
        \mathrm{Gini}_{\leq 23.5} &= 1 - \left[\left(\tfrac{0}{1}\right)^2 + \left(\tfrac{1}{1}\right)^2\right] = 0 \\
        \mathrm{Gini}_{> 23.5} &= 1 - \left[\left(\tfrac{3}{4}\right)^2 + \left(\tfrac{1}{4}\right)^2\right] = 0.375 \\
        \mathrm{Gini}_{\text{Edad, 23.5}} &= \tfrac{1}{5}\times0 + \tfrac{4}{5}\times0.375 = 0.3
    \end{align*}
\end{frame}

\begin{frame}{Características de CART}
    \begin{itemize}
        \item \textbf{Algoritmo greedy}: Toma la mejor decisión local en cada paso
        \item \textbf{División binaria}: Cada nodo se divide en exactamente dos ramas
        \item \textbf{Índice de Gini}: Minimiza la impureza en cada división
        \item \textbf{Árbol óptimo}: CART garantiza el árbol con menos nodos
    \end{itemize}
\end{frame}

\begin{frame}{Ventajas de CART}
    \begin{itemize}
        \item Fácil de interpretar y visualizar
        \item Maneja variables categóricas y numéricas
        \item No requiere normalización de datos
        \item Maneja valores faltantes automáticamente
        \item No es sensible a valores atípicos
    \end{itemize}
\end{frame}

\begin{frame}{Desventajas de CART}
    \begin{itemize}
        \item Propenso al sobreajuste (overfitting)
        \item Inestable con pequeños cambios en los datos
        \item Sesgado hacia variables con muchos niveles
        \item No funciona bien con relaciones no lineales complejas
        \item Algoritmo greedy puede no encontrar la solución óptima global
    \end{itemize}
\end{frame}

\begin{frame}{Consideraciones Prácticas}
    \begin{itemize}
        \item \textbf{Pruning}: Técnica para reducir el sobreajuste eliminando ramas poco importantes
        \item \textbf{Validación cruzada}: Para evaluar el rendimiento real del modelo
        \item \textbf{Ensemble methods}: Combinar múltiples árboles (Random Forest, Gradient Boosting) para mejorar el rendimiento
        \item \textbf{Balance de clases}: Considerar técnicas de balanceo cuando las clases están desbalanceadas
        \item \textbf{Selección de hiperparámetros}: Ajustar parámetros como profundidad máxima y número mínimo de muestras por hoja
    \end{itemize}
\end{frame}
