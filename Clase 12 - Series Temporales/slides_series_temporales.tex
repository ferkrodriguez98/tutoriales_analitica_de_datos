\begin{frame}{¿Qué es una Serie Temporal?}
    \textbf{Definición:}
    \begin{itemize}
        \item Secuencia de observaciones de una variable medidas a lo largo del tiempo
        \item Intervalos regulares (día, semana, mes, año)
        \item El orden de las observaciones importa
        \item Existe dependencia temporal entre ellas
    \end{itemize}
    
    \vspace{0.5cm}
    
    \textbf{Diferencia clave:}
    \begin{itemize}
        \item En datos tradicionales: cada observación es independiente
        \item En series temporales: el valor en $t$ depende de valores anteriores
    \end{itemize}
\end{frame}

\begin{frame}{Ejemplos de Series Temporales}
    \begin{itemize}
        \item \textbf{Finanzas:} Precio de acciones, tipo de cambio, índices bursátiles
        \item \textbf{Economía:} PBI trimestral, tasa de desempleo, inflación mensual
        \item \textbf{Retail:} Ventas diarias de productos, demanda de servicios
        \item \textbf{Clima:} Temperatura diaria, precipitaciones mensuales
        \item \textbf{Tecnología:} Tráfico web, uso de CPU, número de usuarios activos
    \end{itemize}
\end{frame}

\begin{frame}{Características Principales}
    \begin{enumerate}
        \item \textbf{Dependencia temporal:} El valor en el tiempo $t$ puede depender de valores en tiempos anteriores $t-1, t-2, \ldots$
        
        \vspace{0.3cm}
        
        \item \textbf{Autocorrelación:} La correlación de una variable consigo misma en diferentes momentos del tiempo
        
        \vspace{0.3cm}
        
        \item \textbf{Orden:} El orden de las observaciones no puede alterarse sin perder información
    \end{enumerate}
\end{frame}

    \begin{columns}

            \begin{itemize}

            \begin{itemize}

            \end{itemize}
        \end{column}
    \end{columns}
    
    \vspace{0.5cm}
    
    \begin{center}
        Ambos enfoques son complementarios
    \end{center}
\end{frame}

\begin{frame}{Componentes de una Serie Temporal}
    Una serie temporal $Y_t$ se puede descomponer en cuatro componentes:
    
    \vspace{0.5cm}
    
    \begin{itemize}
        \item \textbf{Tendencia ($T_t$):} Comportamiento de largo plazo (crecimiento o decrecimiento sostenido)
        \item \textbf{Ciclo ($C_t$):} Fluctuaciones de mediano plazo sin frecuencia fija
        \item \textbf{Estacionalidad ($S_t$):} Patrones que se repiten con periodicidad fija
        \item \textbf{Ruido ($\epsilon_t$):} Componente aleatoria impredecible
    \end{itemize}
\end{frame}

\begin{frame}{Modelos de Descomposición}
    \textbf{Modelo Aditivo:}
    \[
    Y_t = T_t + S_t + \epsilon_t
    \]
    
    Variaciones estacionales constantes a lo largo del tiempo
    
    \vspace{0.3cm}
    
    \small{Ejemplo: Si las ventas aumentan siempre en 1000 unidades en diciembre, independientemente de si el nivel base es 5000 o 50000 unidades}
    
    \vspace{0.5cm}
    
    \textbf{Modelo Multiplicativo:}
    \[
    Y_t = T_t \times S_t \times \epsilon_t
    \]
    
    Variaciones estacionales proporcionales al nivel de la serie
    
    \vspace{0.3cm}
    
    \small{Ejemplo: Si las ventas aumentan un 20\% en diciembre, la cantidad absoluta depende del nivel base: 1000 unidades extra si el nivel es 5000, pero 10000 unidades extra si el nivel es 50000}
\end{frame}

\begin{frame}{Media Móvil Simple (SMA)}
    \begin{center}
        \Large{\textbf{Fórmula:}}
        
        \vspace{0.5cm}
        
        \[
        \text{SMA}_k(t) = \frac{1}{k} \sum_{i=0}^{k-1} Y_{t-i} = \frac{Y_t + Y_{t-1} + \cdots + Y_{t-k+1}}{k}
        \]
        
        \vspace{0.8cm}
        
        \textbf{Predicción:}
        \[
        \hat{Y}_{t+1} = \text{SMA}_k(t)
        \]
    \end{center}
\end{frame}

\begin{frame}{Limitación de la Media Móvil Simple}
    \textbf{Importante:} Para predecir más de un paso adelante, todas las predicciones usan el mismo valor:
    
    \[
    \hat{Y}_{t+h} = \text{SMA}_k(t) \quad \text{para cualquier } h \geq 1
    \]
    
    \vspace{0.5cm}
    
    \begin{itemize}
        \item Solo útil para predicciones a muy corto plazo (1-2 pasos)
        \item No captura tendencia ni estacionalidad en predicciones futuras
    \end{itemize}
\end{frame}

\begin{frame}{Elección de $k$}
    \begin{columns}
        \begin{column}{0.48\textwidth}
            \textbf{$k$ pequeño (3-5):}
            \begin{itemize}
                \item Responde rápido a cambios
                \item Más sensible al ruido
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{$k$ grande (10-20):}
            \begin{itemize}
                \item Suaviza más
                \item Responde lento a cambios reales
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Media Móvil Ponderada (WMA)}
    \begin{center}
        \Large{\textbf{Fórmula:}}
        
        \vspace{0.5cm}
        
        \[
        \text{WMA}_k(t) = \sum_{i=0}^{k-1} w_i \cdot Y_{t-i}
        \]
        
        donde $\sum w_i = 1$ y típicamente $w_0 > w_1 > \cdots > w_{k-1}$
        
        \vspace{0.8cm}
        
        \textbf{Ejemplo para $k=3$:} $w_0 = 0.5, w_1 = 0.3, w_2 = 0.2$
        
        \[
        \text{WMA}_3(10) = 0.5 \times 33 + 0.3 \times 27 + 0.2 \times 29 = 30.4
        \]
    \end{center}
\end{frame}

\begin{frame}{Suavizamiento Exponencial Simple}
    \begin{center}
        \Large{\textbf{Fórmula:}}
        
        \[
        \hat{Y}_{t+1} = \alpha Y_t + (1-\alpha) \hat{Y}_t
        \]
        
        donde $\alpha \in [0,1]$ es el parámetro de suavizamiento
        
        \vspace{0.5cm}
        
        \begin{itemize}
            \item $\alpha$ cercano a 1: más peso a observaciones recientes
            \item $\alpha$ cercano a 0: más peso a valores históricos
        \end{itemize}
    \end{center}
\end{frame}

\begin{frame}{Método de Holt (Tendencia Lineal)}
    \textbf{Componentes:}
    \begin{itemize}
        \item \textbf{Nivel ($L_t$):} Valor base estimado de la serie
        \item \textbf{Tendencia ($T_t$):} Cambio (pendiente) del nivel
    \end{itemize}
    
    \vspace{0.5cm}
    
    \textbf{Ecuaciones:}
    \begin{align*}
    L_t &= \alpha Y_t + (1-\alpha)(L_{t-1} + T_{t-1}) \\
    T_t &= \beta(L_t - L_{t-1}) + (1-\beta)T_{t-1} \\
    \hat{Y}_{t+h} &= L_t + h \cdot T_t
    \end{align*}
    
    donde $\alpha$ controla el nivel y $\beta$ controla la tendencia
\end{frame}

\begin{frame}{Holt-Winters: Modelo Aditivo}
    \begin{center}
        \textbf{Cuando la estacionalidad es constante:}
        
        \vspace{0.3cm}
        
        \small
        \begin{align*}
        L_t &= \alpha (Y_t - S_{t-s}) + (1-\alpha)(L_{t-1} + T_{t-1}) \\
        T_t &= \beta(L_t - L_{t-1}) + (1-\beta)T_{t-1} \\
        S_t &= \gamma(Y_t - L_t) + (1-\gamma)S_{t-s} \\
        \hat{Y}_{t+h} &= L_t + h \cdot T_t + S_{t+h-s}
        \end{align*}
        
        \vspace{0.3cm}
        
        donde $s$ es la longitud del período estacional (ej: 12 para datos mensuales)
    \end{center}
\end{frame}

\begin{frame}{Holt-Winters: Modelo Multiplicativo}
    \begin{center}
        \textbf{Cuando la estacionalidad crece con el nivel:}
        
        \vspace{0.3cm}
        
        \small
        \begin{align*}
        L_t &= \alpha \frac{Y_t}{S_{t-s}} + (1-\alpha)(L_{t-1} + T_{t-1}) \\
        T_t &= \beta(L_t - L_{t-1}) + (1-\beta)T_{t-1} \\
        S_t &= \gamma\frac{Y_t}{L_t} + (1-\gamma)S_{t-s} \\
        \hat{Y}_{t+h} &= (L_t + h \cdot T_t) \times S_{t+h-s}
        \end{align*}
    \end{center}
\end{frame}

\begin{frame}{Interpretación de los Parámetros}
    \textbf{¿De dónde salen $\alpha$, $\beta$ y $\gamma$?}
    
    Se calculan automáticamente al ajustar el modelo. El software (por ejemplo, statsmodels en Python) encuentra los valores que minimizan el error de predicción en los datos históricos.
    
    \textbf{Interpretación:}
    \begin{itemize}
        \item \textbf{$\alpha$ (nivel):} Qué tan rápido se adapta al valor actual. Valores altos (0.7-0.9) reaccionan rápido a cambios
        \item \textbf{$\beta$ (tendencia):} Qué tan rápido se adapta a cambios en la tendencia. Valores bajos (0.1-0.3) evitan fluctuaciones
        \item \textbf{$\gamma$ (estacionalidad):} Qué tan rápido se adapta a cambios en patrones estacionales. Valores bajos (0.1-0.2) mantienen patrones estables
    \end{itemize}
\end{frame}

\begin{frame}{Cuándo usar Holt-Winters}
    \begin{itemize}
        \item Series con tendencia y estacionalidad bien definidas
        \item Datos mensuales, trimestrales o semanales con patrones repetitivos
        \item Pronósticos a corto o mediano plazo (hasta uno o dos ciclos estacionales)
        \item Sectores como retail, turismo o energía con patrones estacionales marcados
    \end{itemize}
\end{frame}

\begin{frame}{Limitaciones de Holt-Winters}
    \begin{itemize}
        \item Supone que los patrones del pasado se mantendrán en el futuro
        \item No considera el impacto de variables externas (promociones, cambios económicos)
        \item La incertidumbre de las predicciones aumenta rápidamente con el horizonte
    \end{itemize}
\end{frame}

\begin{frame}{ARIMA}
    \begin{center}
        \Large{\textbf{AutoRegressive Integrated Moving Average}}
        
        \vspace{0.8cm}
        
        Familia de modelos estadísticos para análisis y predicción de series temporales
        
        \vspace{0.5cm}
        
        Más sofisticado que métodos anteriores
        
        \vspace{0.5cm}
        
        Puede capturar una mayor variedad de patrones
    \end{center}
\end{frame}

\begin{frame}{Estacionaridad}
    \textbf{Definición:} Una serie es estacionaria cuando sus propiedades estadísticas (media, varianza) no cambian con el tiempo
    
    \vspace{0.5cm}
    
    \textbf{Señales de no estacionaridad:}
    \begin{itemize}
        \item Tendencia clara (creciente o decreciente)
        \item Varianza que cambia con el tiempo
        \item Estacionalidad fuerte
    \end{itemize}
\end{frame}

\begin{frame}{Diferenciación}
    \begin{center}
        \textbf{Primera diferencia:}
        \[
        \Delta Y_t = Y_t - Y_{t-1}
        \]
        
        \textbf{Segunda diferencia:}
        \[
        \Delta^2 Y_t = \Delta Y_t - \Delta Y_{t-1} = (Y_t - Y_{t-1}) - (Y_{t-1} - Y_{t-2})
        \]
    \end{center}
    
    \vspace{0.3cm}
    
    \textbf{¿Cuántas diferencias usar?}
    \begin{itemize}
        \item $d=1$: cuando la serie tiene tendencia lineal
        \item $d=2$: cuando después de la primera diferencia todavía hay tendencia
    \end{itemize}
\end{frame}

\begin{frame}{AR(p) - AutoRegresivo}
    \begin{center}
        \Large{\textbf{Componente Autoregresivo}}
        
        \[
        Y_t = c + \phi_1 Y_{t-1} + \phi_2 Y_{t-2} + \cdots + \phi_p Y_{t-p} + \epsilon_t
        \]
        
        El valor actual depende de los últimos $p$ valores pasados
        
        \vspace{0.3cm}
        
        Los coeficientes $\phi_1, \ldots, \phi_p$ se calculan automáticamente al ajustar el modelo
    \end{center}
\end{frame}

\begin{frame}{I(d) - Integrado}
    \begin{center}
        \Large{\textbf{Componente de Integración}}
        
        Indica cuántas veces diferenciamos la serie para hacerla estacionaria:
        
        
        \begin{itemize}
            \item $d=0$: La serie ya es estacionaria
            \item $d=1$: Aplicamos primera diferencia
            \item $d=2$: Aplicamos segunda diferencia (raro en la práctica)
        \end{itemize}
    \end{center}
\end{frame}

\begin{frame}{MA(q) - Media Móvil}
    \begin{center}
        \Large{\textbf{Componente de Media Móvil}}
        
        \[
        Y_t = \mu + \epsilon_t + \theta_1 \epsilon_{t-1} + \theta_2 \epsilon_{t-2} + \cdots + \theta_q \epsilon_{t-q}
        \]
        
        El valor actual depende de shocks o errores aleatorios de los últimos $q$ períodos
        
        \vspace{0.3cm}
        
        Los coeficientes $\theta_1, \ldots, \theta_q$ se calculan automáticamente
    \end{center}
\end{frame}

\begin{frame}{Modelo ARIMA(p,d,q) Completo}
    \begin{center}
        \Large{\textbf{Combinando todo:}}
        
        \[
        \Delta^d Y_t = c + \sum_{i=1}^{p} \phi_i \Delta^d Y_{t-i} + \sum_{j=1}^{q} \theta_j \epsilon_{t-j} + \epsilon_t
        \]
        
        \vspace{0.8cm}
        
        Definido sobre la serie diferenciada $d$ veces
    \end{center}
\end{frame}

\begin{frame}{Determinar $d$ (diferenciación)}
    \begin{itemize}
        \item \textbf{Visualizar la serie:} Si tiene tendencia, probablemente necesite $d=1$ o $d=2$
        \item \textbf{Pruebas estadísticas:} Tests como ADF (Augmented Dickey-Fuller) o KPSS indican si la serie es estacionaria
        \item \textbf{Regla práctica:} Empezar con $d=1$ y verificar si la serie diferenciada parece estacionaria
    \end{itemize}
\end{frame}

\begin{frame}{ACF y PACF}
    \textbf{¿Qué son?}
    
    Son métricas que miden la dependencia temporal de una serie. Nos ayudan a entender cómo se relaciona cada valor con sus valores pasados.
    
    \textbf{ACF (Autocorrelation Function):}
    \begin{itemize}
        \item Muestra la correlación entre $Y_t$ y $Y_{t-k}$ para diferentes lags $k$
        \item Si decae lentamente, sugiere diferenciación o términos MA
    \end{itemize}
    
    \vspace{0.3cm}
    
    \textbf{PACF (Partial Autocorrelation Function):}
    \begin{itemize}
        \item Muestra la correlación directa entre $Y_t$ y $Y_{t-k}$ controlando por valores intermedios
        \item Picos significativos sugieren términos AR
    \end{itemize}
\end{frame}

\begin{frame}{Criterios de Información}
    \textbf{¿Qué son?}
    
    Son métricas que nos permiten comparar diferentes modelos ARIMA. Balancean el ajuste del modelo (qué tan bien explica los datos) con su complejidad (número de parámetros). Nos ayudan a elegir el mejor modelo entre varias opciones.
    
    \vspace{0.3cm}
    
    \textbf{AIC (Akaike Information Criterion):}
    \begin{itemize}
        \item Penaliza modelos complejos pero menos que BIC
        \item Útil para comparar modelos con diferentes números de parámetros
        \item Menor AIC indica mejor modelo
    \end{itemize}
    
    \vspace{0.3cm}
    
    \textbf{BIC (Bayesian Information Criterion):}
    \begin{itemize}
        \item Penaliza más la complejidad que AIC
        \item Favorece modelos más simples
        \item También menor es mejor
    \end{itemize}
\end{frame}

\begin{frame}{Auto ARIMA}
    \begin{itemize}
        \item Implementado en Python con \texttt{pmdarima}
        \item Prueba automáticamente diferentes combinaciones de $p$, $d$ y $q$
        \item Selecciona el modelo con mejor AIC
        \item Muy útil cuando no estamos seguros de los valores
        \item Es importante entender qué está haciendo para interpretar los resultados
    \end{itemize}
\end{frame}

\begin{frame}{Métricas de Evaluación}
    \begin{itemize}
        \item \textbf{MAE (Mean Absolute Error):}
        \[
        \text{MAE} = \frac{1}{n} \sum_{i=1}^{n} |Y_i - \hat{Y}_i|
        \]
        \item \textbf{MSE (Mean Squared Error):}
        \[
        \text{MSE} = \frac{1}{n} \sum_{i=1}^{n} (Y_i - \hat{Y}_i)^2
        \]
        \item \textbf{RMSE (Root Mean Squared Error):}
        \[
        \text{RMSE} = \sqrt{\frac{1}{n} \sum_{i=1}^{n} (Y_i - \hat{Y}_i)^2}
        \]
        \item \textbf{MAPE (Mean Absolute Percentage Error):}
        \[
        \text{MAPE} = \frac{100\%}{n} \sum_{i=1}^{n} \left|\frac{Y_i - \hat{Y}_i}{Y_i}\right|
        \]
    \end{itemize}
\end{frame}

\begin{frame}{Cross-Validation para Series Temporales}
    \begin{center}
        \Large{\textbf{¡IMPORTANTE!}}
        
        \vspace{0.8cm}
        
        \textbf{NO podemos usar cross-validation tradicional}
        
        \vspace{0.5cm}
        
        Mezclar los datos violaría la dependencia temporal. Estamos con información del futuro alimentando al pasado.
        
        \vspace{0.5cm}
        
        Usamos \textbf{Time Series Cross-Validation}
    \end{center}
\end{frame}

\begin{frame}{Rolling Window (Ventana Móvil)}
    \begin{center}
        \textbf{El tamaño del train se mantiene constante}
        
        En cada fold, ``deslizamos'' la ventana hacia adelante
        
        \vspace{0.5cm}
        
        \begin{itemize}
            \item \textbf{Fold 1:} Entrenar con meses 1-6, probar con meses 7-8
            \item \textbf{Fold 2:} Entrenar con meses 3-8, probar con meses 9-10
            \item \textbf{Fold 3:} Entrenar con meses 5-10, probar con meses 11-12
        \end{itemize}
        
        \vspace{0.5cm}
        
        El tamaño del entrenamiento siempre es 6 meses
    \end{center}
\end{frame}

\begin{frame}{Expanding Window (Ventana Expansiva)}
    \begin{center}
        \textbf{El tamaño del train crece en cada fold}
        
        El inicio se mantiene fijo y solo avanzamos el final
        
        \vspace{0.5cm}
        
        \begin{itemize}
            \item \textbf{Fold 1:} Entrenar con meses 1-4, probar con meses 5-6
            \item \textbf{Fold 2:} Entrenar con meses 1-6, probar con meses 7-8
            \item \textbf{Fold 3:} Entrenar con meses 1-8, probar con meses 9-10
            \item \textbf{Fold 4:} Entrenar con meses 1-10, probar con meses 11-12
        \end{itemize}
        
        \vspace{0.5cm}
        
        Siempre empezamos desde el mes 1
    \end{center}
\end{frame}

\begin{frame}{¿Cuándo usar cada una?}
    \begin{itemize}
        \item \textbf{Rolling Window:} Útil cuando queremos evaluar el desempeño con una cantidad fija de datos históricos. Simula mejor el escenario real donde tenemos una ventana de tiempo limitada
        \item \textbf{Expanding Window:} Útil cuando queremos aprovechar toda la información histórica disponible. Simula mejor el caso donde acumulamos datos a lo largo del tiempo
    \end{itemize}
\end{frame}

from sklearn.metrics import mean_absolute_error, mean_squared_error
import numpy as np

# Configurar cross-validation temporal

tscv = TimeSeriesSplit(n_splits=5)

mae_scores = []

    \end{lstlisting}

\begin{frame}[fragile]{TimeSeriesSplit: Loop}
    \begin{lstlisting}[numbers=left, numbersep=5pt]
for fold, (train_index, test_index) in enumerate(tscv.split(ts)):
    train = ts.iloc[train_index]
    test = ts.iloc[test_index]
    
    print(f"Fold {fold+1}: Train size={len(train)}, Test size={len(test)}")

    # Entrenar modelo (ejemplo con ARIMA)
    modelo = ARIMA(train, order=(1,1,1))
    fit = modelo.fit()

    # Predecir
    predicciones = fit.forecast(steps=len(test))
    \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{TimeSeriesSplit: Métricas}

    rmse = np.sqrt(mean_squared_error(test, predicciones))

    mae_scores.append(mae)
    rmse_scores.append(rmse)
    print(f"  MAE: {mae:.2f}, RMSE: {rmse:.2f}")

print(f"\nMAE promedio: {np.mean(mae_scores):.2f} (+/- {np.std(mae_scores):.2f})")

\begin{frame}[fragile]{Expanding Window: Implementación Manual}
    \begin{lstlisting}[numbers=left, numbersep=5pt]
# Expanding window: el train crece en cada iteracion
train_size_initial = int(len(ts) * 0.3)  # Empezar con 30% de los datos
n_folds = 5

mae_scores = []
rmse_scores = []

for i in range(n_folds):
    # El train crece en cada iteracion
    train_end = train_size_initial + i * (len(ts) - train_size_initial) // n_folds
    test_start = train_end
    test_end = train_end + (len(ts) - train_size_initial) // n_folds
    
    train = ts[:train_end]
    test = ts[test_start:test_end]
    \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Expanding Window: Evaluación}
    \begin{lstlisting}[numbers=left, numbersep=5pt]
    # Entrenar y evaluar modelo
    modelo = ARIMA(train, order=(1,1,1))
    fit = modelo.fit()
    predicciones = fit.forecast(steps=len(test))
    
    mae = mean_absolute_error(test, predicciones)
    rmse = np.sqrt(mean_squared_error(test, predicciones))
    
    mae_scores.append(mae)
    rmse_scores.append(rmse)
    \end{lstlisting}
\end{frame}
