\begin{frame}{¿Qué es Data Profiling?}
    \begin{center}
        \textbf{Técnica fundamental en el análisis de datos}
    \end{center}
    
    \vspace{0.5cm}
    
    \begin{itemize}
        \item<2->[{\makebox[1em][c]{\textcolor{black}{\faSearch}}}] Entender la estructura de los datos
        \item<3->[{\makebox[1em][c]{\textcolor{black}{\faCheck}}}] Evaluar la calidad del dataset
        \item<4->[{\makebox[1em][c]{\textcolor{black}{\faExclamationTriangle}}}] Identificar problemas potenciales
        \item<5->[{\makebox[1em][c]{\textcolor{black}{\faPlay}}}] Primer paso en cualquier proyecto de analítica o ciencia de datos
    \end{itemize}
\end{frame}

\begin{frame}{Tipos de Variables}

    \onslide<1->{
    \textbf{Variables Numéricas:}
    \begin{itemize}
        \item \textbf{Discretas:} Valores enteros específicos
        \item \textbf{Continuas:} Cualquier valor en un rango
    \end{itemize}
    }
    
    \vspace{0.3cm}
    
    \onslide<2->{
    \textbf{Variables Categóricas:}
    \begin{itemize}
        \item \textbf{Nominales:} Sin orden inherente
        \item \textbf{Ordinales:} Con orden natural
    \end{itemize}
    }
    
    \vspace{0.3cm}
    
    \onslide<3->{
    \textbf{Otros Tipos:}
    \begin{itemize}
        \item \textbf{Fecha y Hora:} Momentos específicos
        \item \textbf{Compuestas:} Múltiples tipos de información
        \begin{itemize}
            \item Ej: Direccion completa (calle, numero, ciudad, pais)
        \end{itemize}
    \end{itemize}
    }
\end{frame}

\begin{frame}{Medidas de Dispersión}
    \begin{center}
        \textbf{¿Qué tan dispersos están los valores?}
    \end{center}
    
    \vspace{0.5cm}
    
    \begin{itemize}
        \item<1-> \textbf{Varianza:} Promedio de desviaciones al cuadrado
        \item<2-> \textbf{Desviación Estándar:} Raíz cuadrada de la varianza
        \item<3-> \textbf{Rango Intercuartil (IQR):} Q3 - Q1
        \item<4-> \textbf{Cuartiles:} División en cuatro partes iguales
    \end{itemize}
\end{frame}

\begin{frame}{Forma de la Distribución}
    \begin{center}
        \textbf{Características de la Distribución}
    \end{center}
    
    \vspace{0.5cm}
    
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Skewness (Asimetría):}
            \begin{itemize}
                \item Mide la simetría de la distribución
                \item Indica si está sesgada hacia la izquierda o derecha
                \item Afecta la elección entre media y mediana
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Curtosis:}
            \begin{itemize}
                \item Mide qué tan puntiaguda o plana es
                \item Compara con la distribución normal
                \item Indica presencia de outliers extremos
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Skewness - Tipos de Distribución}
    
    \begin{center}
        \hspace{-1cm}
        \begin{tikzpicture}
        \begin{axis}[
            width=10cm,
            height=5cm,
            xlabel={x},
            ylabel={f(x)},
            legend style={at={(0.5,1.1)}, anchor=south},
            domain=-3:3,
            samples=400,
            ymin=0,
            ymax=0.5
        ]
        
        % Distribución Normal (simétrica)
        \addplot[blue, thick] {exp(-(x-0)^2/2)/sqrt(2*pi)};
        \addlegendentry{Simétrica (skewness = 0)}
        
        % Skewness positivo
        \addplot[red, thick, domain=-3:3] {
            0.8 * (x+3) * exp(-(x+3)/1.2)
        };
        \addlegendentry{Skewness Positivo}
        
        % Skewness negativo
        \addplot[green!60!black, thick, domain=-3:3] {
            0.8 * (3-x) * exp(-(3-x)/1.2)
        };
        \addlegendentry{Skewness Negativo}
        
        \end{axis}
        \end{tikzpicture}
    \end{center}
\end{frame}

\begin{frame}{Skewness - Interpretación}
    \begin{center}
        \textbf{¿Qué significa cada tipo de asimetría?}
    \end{center}
    
    \vspace{0.5cm}
    
    \begin{itemize}
        \item<1->[{\makebox[1em][c]{\textcolor{black}{\faBalanceScale}}}] \textbf{Skewness = 0:} Distribución simétrica (altura, peso en población general)
        \item<2->[{\makebox[1em][c]{\textcolor{black}{\faArrowRight}}}] \textbf{Skewness $>$ 0:} Cola larga a la derecha (ingresos, precios de viviendas)
        \item<3->[{\makebox[1em][c]{\textcolor{black}{\faArrowLeft}}}] \textbf{Skewness $<$ 0:} Cola larga a la izquierda (edad de jubilación, calificaciones altas)
    \end{itemize}
    
\end{frame}

\begin{frame}{Skewness - Implicaciones Prácticas}
    \begin{center}
        \textbf{¿Qué hacer con distribuciones sesgadas?}
    \end{center}
    
    \vspace{0.5cm}
    
    \begin{itemize}
        \item<1->[{\makebox[1em][c]{\textcolor{black}{\faSort}}}] \textbf{Usar mediana en lugar de media} - Menos afectada por valores extremos
        \item<2->[{\makebox[1em][c]{\textcolor{black}{\faCogs}}}] \textbf{Aplicar transformaciones} - Logaritmo, raíz cuadrada, Box-Cox
        \item<3->[{\makebox[1em][c]{\textcolor{black}{\faExclamationTriangle}}}] \textbf{Revisar algoritmos} - Algunos asumen normalidad
        \item<4->[{\makebox[1em][c]{\textcolor{black}{\faSearch}}}] \textbf{Identificar outliers} - Más comunes en una dirección específica
    \end{itemize}
    
\end{frame}

\begin{frame}{Skewness - Reglas de Interpretación}
    \begin{center}
        \textbf{¿Cuándo considerar una distribución sesgada?}
    \end{center}
    
    \vspace{0.5cm}
    
    \begin{center}
        \begin{tabular}{|p{0.25\textwidth}|p{0.30\textwidth}|p{0.30\textwidth}|}
        \hline
        \textbf{Rango} & \textbf{Interpretación} & \textbf{Acción} \\
        \hline
        \textbf{-0.5 a 0.5} & Aproximadamente simétrica & Usar media normalmente \\
        \hline
        \textbf{-1 a -0.5} & Moderadamente sesgada & Considerar mediana \\
        \textbf{0.5 a 1} & hacia la izquierda/derecha & \\
        \hline
        \textbf{$<$ -1 o $>$ 1} & Altamente sesgada & Usar mediana, transformar \\
        \hline
        \end{tabular}
    \end{center}
\end{frame}

\begin{frame}{Curtosis - Forma de la Distribución}
    \begin{center}
        \hspace{-1cm}
        \begin{tikzpicture}
        \begin{axis}[
            width=10cm,
            height=5cm,
            xlabel={x},
            ylabel={f(x)},
            legend style={at={(0.5,1.1)}, anchor=south},
            domain=-3:3,
            samples=100
        ]
        \addplot[blue, thick] {exp(-x^2/2)/sqrt(2*pi)};
        \addlegendentry{Mesocúrtica (Curtosis = 0)}
        
        \addplot[red, thick] {exp(-x^2/0.5)/sqrt(2*pi*0.5)};
        \addlegendentry{Leptocúrtica (Curtosis > 0)}
        
        \addplot[green, thick] {exp(-x^2/3)/sqrt(2*pi*3)};
        \addlegendentry{Platicúrtica (Curtosis < 0)}
        \end{axis}
        \end{tikzpicture}
    \end{center}
\end{frame}

\begin{frame}{Curtosis - Interpretación}
    \begin{center}
        \textbf{¿Qué significa cada tipo de curtosis?}
    \end{center}
    
    \vspace{0.5cm}
    
    \begin{itemize}
        \item<1->[{\makebox[1em][c]{\textcolor{black}{\faBell}}}] \textbf{Curtosis = 0:} Distribución normal estándar (altura, peso bien controlados)
        \item<2->[{\makebox[1em][c]{\textcolor{black}{\faExclamationTriangle}}}] \textbf{Curtosis $>$ 0:} Más puntiaguda, colas pesadas (retornos financieros, precios de commodities)
        \item<3->[{\makebox[1em][c]{\textcolor{black}{\faMinus}}}] \textbf{Curtosis $<$ 0:} Más plana, colas ligeras (temperaturas diarias, velocidades de viento)
    \end{itemize}
    
\end{frame}

\begin{frame}{Curtosis - Implicaciones Prácticas}
    \begin{center}
        \textbf{¿Qué hacer con diferentes tipos de curtosis?}
    \end{center}
    
    \vspace{0.5cm}
    
    \begin{itemize}
        \item<1->[{\makebox[1em][c]{\textcolor{black}{\faExclamationTriangle}}}] \textbf{Leptocúrtica:} Mayor riesgo de eventos extremos, más outliers
        \item<2->[{\makebox[1em][c]{\textcolor{black}{\faArrowsH}}}] \textbf{Platicúrtica:} Datos más dispersos, menos concentración central
        \item<3->[{\makebox[1em][c]{\textcolor{black}{\faCogs}}}] \textbf{Revisar algoritmos} - Algunos fallan con curtosis extrema
        \item<4->[{\makebox[1em][c]{\textcolor{black}{\faShield}}}] \textbf{Evaluar robustez} - Métodos robustos vs sensibles a outliers
    \end{itemize}
    
\end{frame}

\begin{frame}{Curtosis - Reglas de Interpretación}
    \begin{center}
        \textbf{¿Cuándo considerar una distribución diferente a la normal?}
    \end{center}
    
    \vspace{0.5cm}
    
    \begin{center}
        \begin{tabular}{|p{0.25\textwidth}|p{0.25\textwidth}|p{0.25\textwidth}|}
        \hline
        \textbf{Rango} & \textbf{Interpretación} & \textbf{Acción} \\
        \hline
        \textbf{-0.5 a 0.5} & Aproximadamente normal & Usar métodos estándar \\
        \hline
        \textbf{-1 a -0.5} & Moderadamente diferente & Revisar robustez \\
        \textbf{0.5 a 1} & a la normal & de métodos \\
        \hline
        \textbf{$<$ -1 o $>$ 1} & Muy diferente a normal & Usar métodos robustos \\
        \hline
        \end{tabular}
    \end{center}
\end{frame}

\begin{frame}{Datos Faltantes - Tipos}
    \begin{center}
        \textbf{Clasificación de Datos Faltantes}
    \end{center}
    
    \vspace{0.5cm}
    
    \begin{itemize}
        \item<1->[{\makebox[1em][c]{\textcolor{black}{\faRandom}}}] \textbf{MCAR (Missing Completely At Random):} Faltantes aleatorios
        \item<2->[{\makebox[1em][c]{\textcolor{black}{\faLink}}}] \textbf{MAR (Missing At Random):} Faltantes relacionados con variables observables
        \item<3->[{\makebox[1em][c]{\textcolor{black}{\faExclamation}}}] \textbf{MNAR (Missing Not At Random):} Faltantes relacionados con el valor faltante
    \end{itemize}
\end{frame}

\begin{frame}{Otros Aspectos Críticos}
    \begin{center}
        \textbf{Elementos importantes a considerar}
    \end{center}
    
    \vspace{0.5cm}
    
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Cardinalidad:}
            \begin{itemize}
                \item Número de valores únicos en variables categóricas
                \item Puede indicar problemas de calidad
            \end{itemize}
            
            \vspace{0.3cm}
            
            \textbf{Etiquetas Raras:}
            \begin{itemize}
                \item Categorías con muy pocas observaciones
                \item Pueden causar problemas en el modelado
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Valores Extremos:}
            \begin{itemize}
                \item Outliers que afectan el análisis
                \item Deben ser identificados y tratados
            \end{itemize}
            
            \vspace{0.3cm}
            
            \textbf{Magnitudes y Escalas:}
            \begin{itemize}
                \item Variables con escalas muy distintas
                \item Pueden afectar el rendimiento de algoritmos
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Supuestos de Linealidad}
    \begin{center}
        \textbf{Para análisis que asumen relaciones lineales}
    \end{center}
    
    \vspace{0.5cm}
    
    \begin{itemize}
        \item<1->[{\makebox[1em][c]{\textcolor{black}{\faLineChart}}}] \textbf{Relación Lineal:} La relación entre variables debe ser aproximadamente lineal
        \item<2->[{\makebox[1em][c]{\textcolor{black}{\faBell}}}] \textbf{Normalidad:} Los residuos deben seguir una distribución normal
        \item<3->[{\makebox[1em][c]{\textcolor{black}{\faChain}}}] \textbf{Ausencia de Colinealidad:} Variables independientes no altamente correlacionadas
        \item<4->[{\makebox[1em][c]{\textcolor{black}{\faArrowsH}}}] \textbf{Homocedasticidad:} Varianza constante de los residuos
    \end{itemize}
\end{frame}
