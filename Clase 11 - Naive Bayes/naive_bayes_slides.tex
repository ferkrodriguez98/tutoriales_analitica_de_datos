\begin{frame}{¿Qué es Naive Bayes?}
    \textbf{Definición:}
    \begin{itemize}
        \item Algoritmo de aprendizaje automático basado en el teorema de Bayes
        \item Asume independencia condicional entre características
        \item ``Ingenuo'' por esta suposición simplificadora
    \end{itemize}
    
    \vspace{0.5cm}
    
    \textbf{¿Por qué funciona bien?}
    \begin{itemize}
        \item Simple y rápido de entrenar
        \item Efectivo en muchas situaciones reales
        \item Maneja bien datos faltantes
    \end{itemize}
\end{frame}

\begin{frame}{Teorema de Bayes}
    \begin{center}
        \Large{\textbf{Fórmula fundamental:}}
        
        \vspace{0.5cm}
        
        \[
        P(Y|X) = \frac{P(X|Y)P(Y)}{P(X)}
        \]
        
        \vspace{0.8cm}
        
        \begin{tabular}{ll}
            $P(Y|X)$ & Probabilidad posterior \\
            $P(X|Y)$ & Verosimilitud \\
            $P(Y)$ & Probabilidad a priori \\
            $P(X)$ & Evidencia
        \end{tabular}
    \end{center}
\end{frame}

\begin{frame}{Suposición de Independencia}
    \begin{center}
        \textbf{Naive Bayes asume que las características son independientes:}
        
        \[
        P(X|Y) = P(X_1|Y) \times P(X_2|Y) \times \cdots \times P(X_n|Y)
        \]
        
        \vspace{0.8cm}
        
        \begin{tikzpicture}[node distance=1.5cm]
            \node (y) [circle, draw, fill=primary!20, minimum size=1cm] {$Y$};
            \node (x1) [circle, draw, fill=secondary!20, minimum size=0.8cm, below left=of y] {$X_1$};
            \node (x2) [circle, draw, fill=secondary!20, minimum size=0.8cm, below=of y] {$X_2$};
            \node (x3) [circle, draw, fill=secondary!20, minimum size=0.8cm, below right=of y] {$X_3$};
            
            \draw[->] (y) -- (x1);
            \draw[->] (y) -- (x2);
            \draw[->] (y) -- (x3);
        \end{tikzpicture}
    \end{center}
\end{frame}

\begin{frame}{Tipos de Naive Bayes}
    \begin{itemize}
        \item \textbf{BernoulliNB}: para variables binarias (presencia/ausencia de una característica).
        \item \textbf{CategoricalNB}: para datos categóricos o discretos.
        \item \textbf{MultinomialNB}: para conteos y frecuencias (por ejemplo, clasificación de texto).
        \item \textbf{ComplementNB}: para situaciones con clases desbalanceadas.
        \item \textbf{GaussianNB}: para variables continuas (asume distribución normal).
    \end{itemize}
\end{frame}

\begin{frame}{Variables Categóricas vs Continuas}
    \begin{center}
        \textbf{¿Cuál es la diferencia?}
    \end{center}

    \vspace{0.5cm}

    \begin{columns}
        \begin{column}{0.48\textwidth}
            \textbf{Variables Categóricas}
            \begin{itemize}
                \item Valores discretos
                \item Ej: Bajo, Medio, Alto
                \item Método: Contar frecuencias
                \item Fórmula simple: $\frac{count}{total}$
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Variables Continuas}
            \begin{itemize}
                \item Valores reales
                \item Ej: 25.3, 70.5, 1.78
                \item Método: Calcular $\mu$ y $\sigma$
                \item Fórmula: Dist. Normal
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Gaussian Naive Bayes}
    \textbf{Para variables continuas:}

    En lugar de contar frecuencias, calculamos dos parámetros por característica y clase:

    \vspace{0.5cm}

    \begin{itemize}
        \item \textbf{Media ($\mu$):} Promedio de los valores
        \item \textbf{Desviación estándar ($\sigma$):} Dispersión de los valores
    \end{itemize}

    \vspace{0.8cm}

    \textbf{Función de densidad de probabilidad gaussiana:}

    \[
    P(x|y) = \frac{1}{\sqrt{2\pi\sigma^2}} \exp\left(-\frac{(x-\mu)^2}{2\sigma^2}\right)
    \]
\end{frame}

\begin{frame}{Gaussian NB: Procedimiento}
    \begin{center}
        \textbf{Pasos para clasificar:}
    \end{center}

    \vspace{0.5cm}

    \begin{enumerate}
        \item Para cada característica y cada clase, calcular $\mu$ y $\sigma$

        \vspace{0.3cm}

        \item Para un nuevo dato, aplicar la fórmula gaussiana usando los parámetros de cada clase

        \vspace{0.3cm}

        \item Multiplicar las probabilidades de todas las características

        \vspace{0.3cm}

        \item Elegir la clase con mayor probabilidad posterior
    \end{enumerate}
\end{frame}

\begin{frame}{Suavizado de Laplace}
    \begin{center}
        \textbf{Problema:} Probabilidades condicionales cero

        \textbf{Solución:} Suavizado de Laplace

        \[
        P(x_i|y) = \frac{count(x_i, y) + \alpha}{count(y) + \alpha \times n}
        \]
    \end{center}

    \textbf{Ejemplo:}
    \begin{itemize}
        \item $\alpha = 1$ (suavizado de Laplace)
        \item $count(x_i, y) = 0$ (palabra nunca apareció)
        \item $count(y) = 10$ (total documentos en clase)
        \item $n = 1000$ (cantidad características)
    \end{itemize}

    \[
    P(x_i|y) = \frac{0 + 1}{10 + 1 \times 1000} = \frac{1}{1010} \approx 0.001
    \]
\end{frame}

\begin{frame}{Ejercicio 2: Probabilidades a Priori}
    \begin{center}
        \textbf{Probabilidades a priori:}
        \begin{align*}
        P(\text{Riesgo Bajo}) &= \frac{4}{15} \approx 0.267 \\
        P(\text{Riesgo Medio}) &= \frac{6}{15} = 0.4 \\
        P(\text{Riesgo Alto}) &= \frac{5}{15} \approx 0.333
        \end{align*}
    \end{center}
\end{frame}

\begin{frame}{Ejercicio 2: ¡Problema con Probabilidad Cero!}
    \begin{center}
        \textbf{Calculando para Riesgo Bajo (sin suavizado):}
    \end{center}

    \begin{align*}
    P(\text{Bajo}|X) &= P(\text{Bajo}) \times P(\text{Medio}|\text{Bajo}) \\
    &\quad \times P(\text{Bueno}|\text{Bajo}) \times P(\text{Temporal}|\text{Bajo}) \\
    &\quad \times P(\text{Bajo deudas}|\text{Bajo}) \\
    &= \frac{4}{15} \times \frac{1}{4} \times \frac{3}{4} \times \frac{0}{4} \times \frac{4}{4} \\
    &= 0
    \end{align*}

    \vspace{0.5cm}

    \begin{center}
        \textcolor{accent}{\textbf{¡La probabilidad es cero!}}

        Ningún cliente de Riesgo Bajo tiene empleo Temporal
    \end{center}
\end{frame}

\begin{frame}{Ejercicio 2: Solución con Laplace Smoothing}
    \textbf{Aplicamos suavizado de Laplace ($\alpha = 1$):}

    \[
    P(x_i|y) = \frac{count(x_i, y) + 1}{count(y) + n}
    \]

    \vspace{0.3cm}

    \textbf{Recalculando probabilidades para Riesgo Bajo:}
    \begin{align*}
    P(\text{Medio}|\text{Bajo}) &= \frac{1 + 1}{4 + 3} = \frac{2}{7} \approx 0.286 \\
    P(\text{Bueno}|\text{Bajo}) &= \frac{3 + 1}{4 + 3} = \frac{4}{7} \approx 0.571 \\
    P(\text{Temporal}|\text{Bajo}) &= \frac{0 + 1}{4 + 2} = \frac{1}{6} \approx 0.167 \\
    P(\text{Bajo deudas}|\text{Bajo}) &= \frac{4 + 1}{4 + 2} = \frac{5}{6} \approx 0.833
    \end{align*}

    \begin{center}
        $P(\text{Bajo}|X) \approx 0.00303$ (ya no es cero!)
    \end{center}
\end{frame}
